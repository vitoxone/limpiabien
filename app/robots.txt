// app/robots.ts
import type { MetadataRoute } from 'next';

export default function robots(): MetadataRoute.Robots {
  const base = 'https://limpiabien.cl';
  return {
    rules: [{
      userAgent: '*',
      allow: ['/', '/servicios', '/servicios/tapices', '/servicios/colchones', '/servicios/alfombras', '/web-para-pymes'],
      disallow: ['/api/', '/admin/', '/cart/', '/_next/'],
    }],
    sitemap: `${base}/sitemap.xml`,
    host: base,
  };
}
